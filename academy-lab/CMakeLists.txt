cmake_minimum_required(VERSION 3.16)
project(academy_cpp_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(academy_exercises src/exercises.cpp)
target_include_directories(academy_exercises PUBLIC include)
if(MSVC)
  target_compile_options(academy_exercises PRIVATE /W4)
else()
  target_compile_options(academy_exercises PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(academy_demo src/main.cpp)
target_link_libraries(academy_demo PRIVATE academy_exercises)
include(CTest)
if(BUILD_TESTING)
  add_executable(academy_tests tests/test_exercises.cpp)
  target_link_libraries(academy_tests PRIVATE academy_exercises)
  foreach(case smoke sum count clamp)
    add_test(NAME ${case} COMMAND academy_tests ${case})
  endforeach()
endif()
